开机默认 → fsm_id=0（零力矩，全身软的）
              │ 需要先 --start
              ↓
          fsm_id=811（运动就绪）
              │ 再 --stand_up
              ↓
          fsm_id=4（站立）← 只有到这一步，rt/arm_sdk 才有机会被混合执行



# 1. 激活运动模式（观察机器人是否绷紧/有反应）
./bin/r1_loco_client --network_interface=enp2s0 --start

# 2. 站立（挂架上会做出站立姿态）
./bin/r1_loco_client --network_interface=enp2s0 --stand_up

# 3. 确认状态已切换（应返回非 0 的稳定值）
./bin/r1_loco_client --network_interface=enp2s0 --get_fsm_id

# 4. 站稳后再跑手臂示例
./bin/r1_arm_sdk_dds_example enp2s0

# 4. 站立 → 阻尼（观察缓慢软下）
./bin/r1_loco_client --network_interface=enp2s0 --damp
